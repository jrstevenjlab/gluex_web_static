# initial build of image from site/
docker build -t halld-offline .

# export image to be used on other machines
docker save halld-offline | gzip > halld-offline.tar.gz

# on other machine:
docker load < halld-offline.tar.gz
docker run -d -p 8080:80 localhost/halld-offline
